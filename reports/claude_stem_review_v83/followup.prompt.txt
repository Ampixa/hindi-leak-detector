def fit(data, rows, selected, vocabulary_size, acoustic, progress):
    torch.manual_seed(82)
    rng = np.random.default_rng(82)
    if not torch.backends.mps.is_available():
        raise RuntimeError('Required k2 MPS unavailable; no device substitution')
    device = 'mps'
    network = (StemVowelNetwork if acoustic else PhoneContextNetwork)(vocabulary_size).to(device)
    phone, scaler = normalize(data['phone'], selected)
    keys = ('mel', 'context', 'labels') + (('targets', 'masks') if acoustic else ())
    arrays = {k: torch.from_numpy(data[k]).to(device) for k in keys}
    arrays['phone'] = torch.from_numpy(phone).to(device)
    current = [rows[i] for i in selected]
    weights = source_weights(current)
    lookup = {int(i): float(w) for i, w in zip(selected, weights, strict=True)}
    # Each auxiliary target class has exactly half the total source-weighted
    # mass. Normalize per row so short nuclei are not downweighted by length.
    classes = np.array([float(data['targets'][i, 0]) for i in selected]) if acoustic else None
    aux_lookup = {}
    if acoustic:
        for c in (0., 1.):
            mass = float(weights[classes == c].sum())
            if mass <= 0:
                raise ValueError('Missing auxiliary target class')
            for i, w, label in zip(selected, weights, classes, strict=True):
                if label == c:
                    aux_lookup[int(i)] = float(w * len(selected) / (2 * mass))
    groups, sources = {}, {}
    for i in selected:
        groups.setdefault(rows[i]['phone_collision_group'], set()).add(rows[i]['pair_id'])
        sources.setdefault(rows[i]['pair_id'], []).append(int(i))
    normalizer = float(np.mean([1 / len(groups[current[a]['phone_collision_group']]) for a, _ in pair_indices(current)]))
    rank = {int(i): 1 / len(groups[rows[i]['phone_collision_group']]) / normalizer for i in selected}
    identities = sorted(sources)
    optimizer = torch.optim.AdamW(network.parameters(), lr=.001, weight_decay=.01, foreach=False)
    history, started = [], time.monotonic()
    for epoch in range(EPOCHS):
        network.train()
        order, losses, auxiliaries = rng.permutation(len(identities)), [], []
        for start in range(0, len(order), 14):
            at = [i for n in order[start:start + 14] for i in sources[identities[n]]]
            ids = torch.tensor(at, device=device)
            optimizer.zero_grad(set_to_none=True)
            inputs = [arrays[k][ids] for k in ('mel', 'context', 'phone')]
            if acoustic:
                logits, dense = network(*inputs, auxiliary=True)
                frame = F.binary_cross_entropy_with_logits(dense, arrays['targets'][ids], reduction='none')
                masked = (frame * arrays['masks'][ids]).sum(1) / arrays['masks'][ids].sum(1)
                aux = (masked * torch.tensor([aux_lookup[i] for i in at], device=device)).mean()
            else:
                logits = network(*inputs)
                aux = torch.tensor(0., device=device)
            w = torch.tensor([lookup[i] for i in at], device=device)
            loss = (F.binary_cross_entropy_with_logits(logits, arrays['labels'][ids], reduction='none') * w).mean()
            loss = loss + .2 * rank_loss(logits, [rows[i] for i in at], [rank[i] for i in at]) + .5 * aux
            if not torch.isfinite(loss):
                raise ValueError('Nonfinite localized acoustic objective')
            loss.backward()
            torch.nn.utils.clip_grad_norm_(network.parameters(), 5., error_if_nonfinite=True)
            optimizer.step()
            losses.append(float(loss.detach().cpu())); auxiliaries.append(float(aux.detach().cpu()))
        item = {'epoch': epoch + 1, 'loss': float(np.mean(losses)), 'auxiliary_loss': float(np.mean(auxiliaries)),
                'elapsed_seconds': time.monotonic() - started}
        if acoustic:
            item['vowel_veto_strength'] = float(F.softplus(network.vowel_veto).detach().cpu())
        history.append(item); progress(item)
    return network.eval().cpu(), scaler, history


def load_targets(data, part, cache, targets):
    report = json.loads((targets / 'report.json').read_text())['partitions'][part]
    if sha(targets / (part + '.npz')) != report['sha256'] or sha(cache / (part + '.index.json')) != report['input_index_sha256']:
        raise ValueError('Loss target/input identity changed')
    with np.load(targets / (part + '.npz'), allow_pickle=False) as z:
        data.update({k: z[k].copy() for k in ('targets', 'masks')})
    if data['targets'].shape != (len(data['labels']), 128) or data['masks'].shape != data['targets'].shape:
        raise ValueError('Loss target/input dimensions differ')
    if not np.isin(data['targets'], [0., 1.]).all() or not np.isfinite(data['masks']).all() or np.any(data['masks'] < 0) or np.any(data['masks'] > 1) or np.any(data['masks'].sum(1) <= 0):
        raise ValueError('Invalid loss-only targets')


def dense_diagnostic(network, scaler, data, indices):
    """OOF only: known-region vowel scores; not an inference input."""
    values, pooled = [], []
    phone = np.clip((data['phone'] - scaler['mean']) / scaler['scale'], -8, 8).astype(np.float32)
    with torch.inference_mode():
        for start in range(0, len(indices), 128):
            at = indices[start:start + 128]
            _, dense = network(torch.from_numpy(data['mel'][at]), torch.from_numpy(data['context'][at]), torch.from_numpy(phone[at]), auxiliary=True)
            mask = torch.from_numpy(data['masks'][at])
            values.extend(((dense * mask).sum(1) / mask.sum(1)).tolist())
            focus = torch.from_numpy(data['mel'][at, 512])
            focus = torch.where(focus >= .25, focus, torch.zeros_like(focus))
            local = (dense + torch.log(focus.clamp_min(1e-30))).masked_fill(focus == 0, -torch.inf)
            pooled.extend((torch.logsumexp(local, 1) - torch.log(focus.sum(1))).tolist())
    return np.array(values), np.array(pooled)

import unittest
import math

import numpy as np
import torch
import torch.nn.functional as F

from hindi_detector.stem_deletion import StemVowelNetwork, interpolate_stem, query_grid
from hindi_detector.nucleus_deletion import PhoneContextNetwork


class StemClockTests(unittest.TestCase):
    def test_grid_matches_actual_centered_stft_and_pytorch_resampling(self):
        for bounds, samples in [((0., .2), 16000), ((.8, 1.), 16000), ((.6687942708333334, 1.1349236111111112), 23347)]:
            left = max(0, math.floor((bounds[0] - .04) * 16000))
            right = min(samples, math.ceil((bounds[1] + .04) * 16000))
            spectrum = torch.stft(torch.zeros(right - left), n_fft=512, hop_length=80,
                win_length=400, window=torch.hann_window(400), center=True, return_complex=True)
            centers = left + torch.arange(spectrum.shape[1], dtype=torch.float64) * 80
            actual = F.interpolate(centers[None, None], size=128, mode='linear', align_corners=False)[0, 0].numpy()
            np.testing.assert_allclose(query_grid(bounds, samples), actual, atol=1e-8, rtol=0)

    def test_phone_context_arm_never_reads_any_audio_channels(self):
        net = PhoneContextNetwork(10).eval()
        context, phone = torch.zeros(2, 2, dtype=torch.long), torch.zeros(2, 14)
        before = net(torch.randn(2, 66, 128), context, phone)
        after = net(torch.randn(2, 514, 128) * 100, context, phone)
        torch.testing.assert_close(before, after)

    def test_actual_conv_centers_resample_to_existing_target_clock(self):
        samples = 24000
        frames = (samples - 160) // 80 + 1
        centers = 79.5 + np.arange(frames, dtype=np.float32) * 80
        raw = np.broadcast_to(centers, (512, frames))
        bounds = (.3, 1.1)
        actual = interpolate_stem(raw, bounds, samples)
        np.testing.assert_allclose(actual[0], query_grid(bounds, samples), atol=.002)
        with self.assertRaises(ValueError):
            interpolate_stem(raw[:, ::4], bounds, samples)

    def test_dense_branch_has_no_word_loss_gradient_or_hint_input(self):
        net = StemVowelNetwork(10).eval()
        values = torch.randn(2, 514, 128)
        values[:, 512:] = .5
        context, phone = torch.zeros(2, 2, dtype=torch.long), torch.zeros(2, 14)
        logits, dense = net(values, context, phone, auxiliary=True)
        logits.sum().backward()
        self.assertTrue(all(p.grad is None for p in net.temporal.parameters()))
        self.assertTrue(all(p.grad is None for p in net.nucleus.parameters()))
        values[:, 512:] = .8
        _, changed = net(values, context, phone, auxiliary=True)
        torch.testing.assert_close(dense, changed)

def supervision(row, samples):
    """Known manipulation coordinates create LOSS targets, never model inputs."""
    start, end = row['production_word']['start'], row['production_word']['end']
    left = max(0, int(np.floor((start - .04) * 16000)))
    right = min(samples, int(np.ceil((end + .04) * 16000)))
    raw_frames = (right - left) // 80 + 1  # Verified centered512 FFT, hop80.
    at = np.clip((np.arange(128) + .5) * raw_frames / 128 - .5, 0, raw_frames - 1)
    times = left / 16000 + at * 80 / 16000
    half = np.maximum(np.gradient(times) / 2, .0025)
    a, b = row['source_region']['start_sample'], row['source_region']['end_sample']
    edit = row['edit']
    if row['family'] in {'P1', 'P1w', 'S2', 'S0', 'S_world'}:
        if (edit['start_sample'], edit['end_sample']) != (a, b):
            raise ValueError('Physical edit and source nucleus coordinates disagree')
    if row['family'] in {'S0', 'S_world'} and edit['replacement_samples'] != b - a:
        raise ValueError('Retained control unexpectedly changes nucleus duration')
    if row['family'] in {'P1', 'P1w'}:
        if edit['replacement_samples'] != 0:
            raise ValueError('Deletion target has retained replacement samples')
        lo, hi, value = edit['start_sample'] / 16000 - .02, edit['start_sample'] / 16000 + .02, 0.
    else:
        value = 1.
        if row['family'] == 'S2':
            a, b = edit['start_sample'], edit['start_sample'] + edit['replacement_samples']
        elif row['family'] == 'S1':
            if edit['end_sample'] <= a:
                shift = edit['replacement_samples'] - (edit['end_sample'] - edit['start_sample'])
                a, b = a + shift, b + shift
            elif edit['start_sample'] < b:
                raise ValueError('Off-target control overlaps the target nucleus')
        elif row['family'] not in {'S_original', 'S0', 'S_world'}:
            raise ValueError('Unregistered auxiliary target family')
        lo, hi = a / 16000, b / 16000
        trim = min(.0125, (hi - lo) / 4)
        lo, hi = lo + trim, hi - trim
    if not 0 <= lo < hi <= samples / 16000:
        raise ValueError('Known physical target outside emitted waveform')
    mask = np.maximum(0., np.minimum(times + half, hi) - np.maximum(times - half, lo)) / (2 * half)
    if mask.sum() <= 0:
        raise ValueError('Auxiliary target has no overlap with observed production word crop')
    return np.full(128, value, np.float32), mask.astype(np.float32)


class PhoneContextNetwork(nn.Module):
    def __init__(self, vocabulary_size):
        super().__init__()
        self.context = nn.Embedding(vocabulary_size, 4)
        self.head = nn.Sequential(nn.Linear(22, 16), nn.SiLU(), nn.Dropout(.1), nn.Linear(16, 1))

    def forward(self, mel, context, phone):
        return self.head(torch.cat([phone, self.context(context).flatten(1)], dim=1)).squeeze(1)


class NucleusDeletionNetwork(PhoneContextNetwork):
    def __init__(self, vocabulary_size):
        super().__init__(vocabulary_size)
        self.temporal = nn.Sequential(nn.Conv1d(64, 24, 5, padding=2), nn.GroupNorm(6, 24), nn.SiLU(),
            nn.Conv1d(24, 24, 3, padding=1), nn.GroupNorm(6, 24), nn.SiLU(),
            nn.Conv1d(24, 24, 3, dilation=2, padding=2), nn.GroupNorm(6, 24), nn.SiLU())
        self.nucleus = nn.Sequential(nn.Conv1d(24, 16, 3, padding=1), nn.GroupNorm(4, 16), nn.SiLU(), nn.Conv1d(16, 1, 1))
        self.vowel_veto = nn.Parameter(torch.tensor(0.))

    def forward(self, mel, context, phone, *, auxiliary=False):
        embedded = self.context(context).flatten(1)
        values = self.temporal(mel[:, :64])
        dense = self.nucleus(values).squeeze(1)
        # Only the observed deletion-anchor neighborhood can supply the acoustic
        # veto. No global acoustic pool and no known edit-location input.
        focus = mel[:, 64]
        focus = torch.where(focus >= .25, focus, torch.zeros_like(focus))
        if torch.any(focus.sum(1) <= 0):
            raise ValueError('No observed local focus support')
        # Mechanistic arm: the word-label loss cannot train the vowel head to
        # become a second deletion classifier. Only local auxiliary labels do.
        local = (dense.detach() + torch.log(focus.clamp_min(1e-30))).masked_fill(focus == 0, -torch.inf)
        present = torch.logsumexp(local, dim=1) - torch.log(focus.sum(1))
        logits = self.head(torch.cat([phone, embedded], dim=1)).squeeze(1) - F.softplus(self.vowel_veto) * present
        return (logits, dense) if auxiliary else logits

def query_grid(bounds, samples):
    start, end = bounds
    left = max(0, int(np.floor((start - .04) * 16000)))
    right = min(samples, int(np.ceil((end + .04) * 16000)))
    count = (right - left) // 80 + 1
    return left + np.clip((np.arange(128) + .5) * count / 128 - .5, 0, count - 1) * 80


def interpolate_stem(features, bounds, samples):
    """Sample actual conv centers onto the immutable mel/target time grid."""
    if features.ndim != 2 or features.shape[0] != 512:
        raise ValueError('Need verified512-channel fifth convolution output')
    # Kernels10,3,3,3,3 and strides5,2,2,2,2: RF160, stride80.
    expected = (samples - 160) // 80 + 1
    if features.shape[1] != expected:
        raise ValueError('Actual early frontend clock differs from pinned convolution recipe')
    at = (query_grid(bounds, samples) - 79.5) / 80
    # Only waveform-edge padding may clamp: the first valid convolution center
    # is4.96875ms into the utterance. No oracle/edit/source location is accepted.
    at = np.clip(at, 0, features.shape[1] - 1)
    lo = np.floor(at).astype(np.int64)
    hi = np.minimum(lo + 1, features.shape[1] - 1)
    fraction = (at - lo).astype(np.float32)
    return features[:, lo] * (1 - fraction) + features[:, hi] * fraction


class StemVowelNetwork(PhoneContextNetwork):
    def __init__(self, vocabulary_size):
        super().__init__(vocabulary_size)
        self.temporal = nn.Sequential(nn.Conv1d(512, 32, 3, padding=1), nn.GroupNorm(8, 32), nn.SiLU(),
            nn.Conv1d(32, 32, 3, padding=1), nn.GroupNorm(8, 32), nn.SiLU())
        self.nucleus = nn.Sequential(nn.Conv1d(32, 16, 3, padding=1), nn.GroupNorm(4, 16), nn.SiLU(), nn.Conv1d(16, 1, 1))
        self.vowel_veto = nn.Parameter(torch.tensor(0.))

    def forward(self, mel, context, phone, *, auxiliary=False):
        # Name retained for the generic fitting adapter; these512channels are
        # pretrained convolution features, not mel bands. Hints excluded.
        dense = self.nucleus(self.temporal(mel[:, :512])).squeeze(1)
        focus = mel[:, 512]
        focus = torch.where(focus >= .25, focus, torch.zeros_like(focus))
        if torch.any(focus.sum(1) <= 0):
            raise ValueError('No observed acoustic focus support')
        values = (dense.detach() + torch.log(focus.clamp_min(1e-30))).masked_fill(focus == 0, -torch.inf)
        present = torch.logsumexp(values, dim=1) - torch.log(focus.sum(1))
        logit = super().forward(mel, context, phone) - F.softplus(self.vowel_veto) * present
        return (logit, dense) if auxiliary else logit

def sha(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()


def audio(pack: Path, row: dict) -> np.ndarray:
    path = (pack / row['audio']).resolve(strict=True)
    if not path.is_relative_to(pack.resolve()) or sha(path) != row['sha256']:
        raise ValueError('ASR input path/checksum differs from paired generation')
    wave, rate = sf.read(path, dtype='float32', always_2d=True)
    if wave.shape[1] != 1 or not np.isfinite(wave).all() or not .2 <= len(wave) / rate <= 30:
        raise ValueError('Need actual finite .2–30s mono TTS for ASR calibration')
    divisor = int(np.gcd(rate, 16000))
    return resample_poly(wave[:, 0], 16000 // divisor, rate // divisor).astype('float32') if rate != 16000 else wave[:, 0]
