"""Reference-free internal-vowel evidence inside fixed consonant anchors.

The deletion path locates both anchors once per boundary recipe. Vowel and
blank-only hypotheses then use the IDENTICAL intervening frames; unrelated
vowels cannot improve that local likelihood. No word length, lexical identity,
speaker identity, source identity or reference-derived feature is emitted.
"""
from __future__ import annotations

import math

import numpy as np

from hindi_detector.exact_phone_ctc import path_score

FEATURE_NAMES = ['gap_deletion_margin_mean', 'gap_deletion_margin_min', 'gap_deletion_margin_max',
    'gap_vowel_mass_mean', 'gap_vowel_peak_mean', 'focal_vowel_log_emission_mean',
    'focal_ownership_mean', 'left_anchor_log_emission_mean', 'right_anchor_log_emission_mean',
    'gap_native_fit_mean', 'gap_blank_fit_mean', 'gap_available_fraction']
FLOOR = -30.


def _paths(compiled, variant, contrast, retained):
    at = retained.index(contrast['index'])
    native = [variant['units'][i] for i in retained]
    native[at] = compiled['any_vowel']
    return native, native[:at] + native[at + 1:], at


def measure(compiled, variant, contrast, crops):
    if contrast['kind'] != 'vowel_deletion' or len(crops) != 3:
        raise ValueError('Anchored evidence is only for deletion at three fixed word bounds')
    index, units, vowels = contrast['index'], variant['units'], compiled['any_vowel']
    if not 0 < index < len(units) - 1 or units[index - 1] & vowels or units[index + 1] & vowels:
        raise ValueError('Internal focal vowel requires two nonsyllabic text-level neighbors')
    retained = list(range(len(units)))
    removable = {i for i, phone in enumerate(variant['phones']) if i != index and phone in {'ax', 'h'}}
    def fit(indices):
        a, b, _ = _paths(compiled, variant, contrast, indices)
        return max(path_score(crops[1][2], a), path_score(crops[1][2], b))
    current = fit(retained)
    while removable:
        trials = []
        for at in sorted(removable):
            proposal = [i for i in retained if i != at]
            # Never remove either immediate consonant anchor around the focal
            # site, even when it happens to be optional /h/ in other positions.
            focal = proposal.index(index)
            if not 0 < focal < len(proposal) - 1 or proposal[focal - 1] != index - 1 or proposal[focal + 1] != index + 1:
                continue
            trials.append((fit(proposal), at, proposal))
        if not trials:
            break
        value, at, proposal = max(trials, key=lambda row: row[0])
        if value <= current + .01:
            break
        retained, current = proposal, value
        removable.remove(at)
    native, alternate, focal = _paths(compiled, variant, contrast, retained)
    measurements = []
    for padding, start_frame, values in crops:
        deletion = path_score(values, alternate, trace=True)
        left, right = deletion['spans'][focal - 1], deletion['spans'][focal]
        start, end = left['end_frame'], right['start_frame']
        if not 0 <= start <= end <= len(values):
            raise ValueError('Consonant anchors do not define an ordered local gap')
        gap = values[start:end]
        mass = peak = 0.
        if len(gap):
            native_score = path_score(gap, [vowels])
            blank_score = path_score(gap, [])
            if not math.isfinite(blank_score):
                raise ValueError('Nonfinite blank-only local hypothesis')
            if not math.isfinite(native_score):
                native_score = FLOOR * len(gap)
            p = np.exp(gap)
            vowel = p[:, sorted(vowels)].sum(1)
            nonblank = p[:, 4:].sum()
            mass = float(vowel.sum() / max(1e-12, nonblank))
            peak = float(vowel.max())
            native_fit, blank_fit = native_score / len(gap), blank_score / len(gap)
        else:
            # A zero-frame observed gap is evidence, not a measurement failure:
            # the two consonant peaks are adjacent. Represent the impossible
            # intervening-vowel hypothesis explicitly with the fixed censor.
            native_fit, blank_fit = FLOOR, 0.
        focal_log, ownership, span = FLOOR, 0., None
        try:
            intact = path_score(values, native, trace=True)
            span = intact['spans'][focal]
            focal_log = span['mean_log_emission']
            overlap = max(0, min(end, span['end_frame']) - max(start, span['start_frame']))
            ownership = overlap / max(1, span['end_frame'] - span['start_frame'])
        except ValueError as error:
            # No intact path is an explicit observed hypothesis outcome, not an
            # omitted positive. The deletion path above must still be feasible.
            if 'No feasible exact CTC path' not in str(error):
                raise
        measurements.append({'padding_seconds': padding, 'start_frame': start_frame,
            'gap_start_frame': start_frame + start, 'gap_end_frame': start_frame + end,
            'gap_deletion_margin': float(np.clip(blank_fit - native_fit, -10., 10.)), 'gap_vowel_mass': mass, 'gap_vowel_peak': peak,
            'native_focal_log_emission': focal_log, 'focal_ownership': ownership,
            'left_anchor': left, 'right_anchor': right, 'native_focal_span': span,
            'gap_native_fit': native_fit, 'gap_blank_fit': blank_fit, 'gap_available': bool(len(gap))})
    def mean(name):
        return float(np.mean([row[name] for row in measurements]))
    margins = [row['gap_deletion_margin'] for row in measurements]
    features = np.array([np.mean(margins), min(margins), max(margins), mean('gap_vowel_mass'), mean('gap_vowel_peak'),
        mean('native_focal_log_emission'), mean('focal_ownership'),
        np.mean([row['left_anchor']['mean_log_emission'] for row in measurements]),
        np.mean([row['right_anchor']['mean_log_emission'] for row in measurements]),
        mean('gap_native_fit'), mean('gap_blank_fit'), mean('gap_available')], np.float32)
    if features.shape != (len(FEATURE_NAMES),) or not np.isfinite(features).all():
        raise ValueError('Invalid anchored-vowel evidence features')
    return features, {'windows': measurements, 'retained_native_indices': retained,
                      'native_target_phones': variant['phones'], 'no_reference_audio_used': True,
                      'likelihood': 'Exact allowed-token CTC inside identical fixed deletion-anchor gaps',
                      'margin_clip': [-10., 10.]}

"""Linear deletion evidence with local features and grouped fitting helpers."""
from __future__ import annotations

from collections import Counter
import hashlib
from pathlib import Path

import numpy as np
import torch
from torch import nn

from hindi_detector.anchored_vowels import FEATURE_NAMES as LOCAL_NAMES
from hindi_detector.corrected_synthesis import sha

# Anchor strength, censored fits and gap-duration availability remain diagnostics.
# The head sees local vowel evidence, not word duration or anchor damage.
INDICES = {'local14': list(range(7)) + list(range(12, 19)), 'primary7': list(range(7)), 'secondary7': list(range(12, 19)),
           'gap_mass4': [3, 4, 15, 16], 'no_margin8': [3, 4, 5, 6, 15, 16, 17, 18]}
ALL_NAMES = ['primary_' + n for n in LOCAL_NAMES] + ['secondary_' + n for n in LOCAL_NAMES]


def select(features, mode):
    values = np.asarray(features, np.float64)
    if values.ndim != 2 or values.shape[1] != 24 or not np.isfinite(values).all() or mode not in INDICES:
        raise ValueError('Expected24 finite anchored deletion features and an explicit mode')
    values = values.copy()
    values[:, [5, 17]] = np.clip(values[:, [5, 17]], -10., 0.)
    return values[:, INDICES[mode]]


def weights(rows):
    """Equal phonetic-group/source/class mass, split over same-source families."""
    per_source_class = Counter((row['pair_id'], row['label']) for row in rows)
    group_sources = {}
    for row in rows:
        group_sources.setdefault(row['phone_collision_group'], set()).add(row['pair_id'])
    result = np.array([1 / (len(group_sources[row['phone_collision_group']]) * per_source_class[row['pair_id'], row['label']]) for row in rows], float)
    return result / result.mean()


def cells(rows, folds=3):
    """Hold out both a phonetic group and a conditioning voice per CV cell."""
    assignments = {}
    for row in rows:
        identity = row['pair_id']
        cell = (int(hashlib.sha256(row['phone_collision_group'].encode()).hexdigest()[:12], 16) % folds, row['voice'])
        if identity in assignments and assignments[identity] != cell:
            raise ValueError('One source family crosses grouped cross-validation cells')
        assignments[identity] = cell
    group_ids = np.array([assignments[row['pair_id']][0] for row in rows])
    voices = np.array([row['voice'] for row in rows])
    for fold in range(folds):
        for voice in sorted(set(voices)):
            validation = (group_ids == fold) & (voices == voice)
            training = (group_ids != fold) & (voices != voice)
            if not validation.any():
                continue
            if training.sum() < 12:
                raise ValueError('Grouped CV cell has insufficient disjoint training rows')
            tr_groups = {row['phone_collision_group'] for row, keep in zip(rows, training, strict=True) if keep}
            dv_groups = {row['phone_collision_group'] for row, keep in zip(rows, validation, strict=True) if keep}
            if tr_groups & dv_groups or set(voices[training]) & set(voices[validation]):
                raise AssertionError('CV phonetic group or voice leakage')
            yield training, validation


def fit(features, labels, rows, mode, ridge):
    values = select(features, mode)
    y = np.asarray(labels, float)
    if len(y) != len(values) or len(rows) != len(y) or set(y.tolist()) != {0., 1.}:
        raise ValueError('Both measured physical labels and matching metadata required')
    if not np.array_equal(y, np.array([row['label'] for row in rows])):
        raise ValueError('Explicit fit labels do not match their source/family metadata')
    mean, scale = values.mean(0), np.maximum(values.std(0), .05)
    x = torch.from_numpy(np.clip((values - mean) / scale, -8, 8))
    target, w = torch.from_numpy(y), torch.from_numpy(weights(rows))
    network = nn.Linear(x.shape[1], 1, dtype=torch.float64)
    nn.init.zeros_(network.weight)
    nn.init.zeros_(network.bias)
    optimizer = torch.optim.LBFGS(network.parameters(), max_iter=120, tolerance_grad=1e-7, tolerance_change=1e-10, line_search_fn='strong_wolfe')
    def closure():
        optimizer.zero_grad(set_to_none=True)
        logits = network(x).squeeze(-1)
        loss = (torch.nn.functional.binary_cross_entropy_with_logits(logits, target, reduction='none') * w).mean()
        loss = loss + ridge * network.weight.square().sum() / 2
        if not torch.isfinite(loss):
            raise ValueError('Nonfinite local-deletion training objective')
        loss.backward()
        return loss
    optimizer.step(closure)
    state = {'schema': 'anchored_deletion_v71', 'feature_mode': mode, 'feature_names': [ALL_NAMES[i] for i in INDICES[mode]],
             'mean': mean, 'scale': scale, 'weights': network.state_dict(), 'ridge': ridge,
             'cause': 'vowel_deletion', 'actual_hindi_error_accuracy': None}
    return network.eval(), state


def score(network, state, features):
    values = select(features, state['feature_mode'])
    normalized = np.clip((values - state['mean']) / state['scale'], -8, 8)
    with torch.inference_mode():
        return torch.sigmoid(network(torch.from_numpy(normalized)).squeeze(-1)).cpu().numpy()


def load(path: Path):
    state = torch.load(path, map_location='cpu', weights_only=False)
    if state['schema'] != 'anchored_deletion_v71' or state['cause'] != 'vowel_deletion':
        raise ValueError('Not a reviewed anchored deletion checkpoint')
    if state['feature_names'] != [ALL_NAMES[i] for i in INDICES[state['feature_mode']]]:
        raise ValueError('Anchored feature schema changed')
    root = Path(__file__).resolve().parents[1]
    for name, digest in state['code_sha256'].items():
        if sha(root / name) != digest:
            raise ValueError('Checkpoint feature-code pin differs: ' + name)
    for name, pin in state['resource_files'].items():
        if sha(root / name) != pin:
            raise ValueError('Checkpoint resource pin differs: ' + name)
    network = nn.Linear(len(state['feature_names']), 1, dtype=torch.float64)
    network.load_state_dict(state['weights'], strict=True)
    return network.eval(), state

def world_versions(samples, start, end):
    """Matched WORLD reconstruction and half-duration focal vowel, pitch held."""
    signal = np.asarray(samples, np.float64)
    f0, times = pyworld.dio(signal, 16000, f0_floor=50., f0_ceil=650., frame_period=5.)
    f0 = pyworld.stonemask(signal, f0, times, 16000)
    spectrum = pyworld.cheaptrick(signal, f0, times, 16000)
    noise = pyworld.d4c(signal, f0, times, 16000)
    original = pyworld.synthesize(f0, spectrum, noise, 16000, frame_period=5.)
    if len(original) < len(samples):
        raise ValueError('WORLD reconstruction shorter than declared input')
    replacement = (end - start) // 2
    target_samples = len(samples) - (end - start) + replacement
    source_knots = np.array([0., start / 16000, end / 16000, len(samples) / 16000])
    target_knots = np.array([0., start / 16000, (start + replacement) / 16000, target_samples / 16000])
    target_times = np.minimum(np.arange(int(np.ceil(target_samples / 80)) + 1) * .005, target_samples / 16000)
    query = np.interp(target_times, target_knots, source_knots)
    query = np.clip(query, times[0], times[-1])
    nearest = np.clip(np.searchsorted(times, query), 1, len(times) - 1)
    nearest -= np.abs(times[nearest - 1] - query) < np.abs(times[nearest] - query)
    sp = np.ascontiguousarray(interp1d(times, spectrum, axis=0)(query))
    ap = np.ascontiguousarray(interp1d(times, noise, axis=0)(query))
    compressed = pyworld.synthesize(np.ascontiguousarray(f0[nearest]), sp, ap, 16000, frame_period=5.)
    if len(compressed) < target_samples:
        raise ValueError('WORLD compression shorter than declared time map')
    return original[:len(samples)].astype(np.float32), compressed[:target_samples].astype(np.float32), replacement

def read_partition(part):
    directory = ROOT / ('reports/native_main_controls_' + part + '_v71')
    content = ROOT / ('reports/native_main_content_' + part + '_v71/result.json')
    report = json.loads((directory / 'report.json').read_text())
    if report['status'] != 'complete_reviewed_local_controls' or sha(directory / 'items.jsonl') != report['items_sha256']:
        raise ValueError('Incomplete/changed partition: ' + part)
    screen = json.loads(content.read_text())
    accepted = set(screen['accepted_ids'])
    screens = {row['id']: row for row in screen['rows']}
    rows, rejected = [], []
    for row in read_rows(directory / 'items.jsonl'):
        if row['split'] != part or row['family'] == 'P1u':
            raise ValueError('Training/development partition contains wrong split/test-only family')
        if row['source_audio_sha256'] != screens[row['job_id']]['sha256']:
            raise ValueError('Content decision belongs to a different source waveform')
        if row['job_id'] not in accepted:
            rejected.append({key: row[key] for key in ('pair_id', 'word', 'voice', 'family', 'label')})
            continue
        if row.get('audio') and sha(directory / row['audio']) != row['audio_sha256']:
            raise ValueError('Training waveform identity differs')
        if row['family'] not in {'P1', 'P1w', 'S0', 'S1', 'S2', 'S_original', 'S_world'}:
            raise ValueError('Unregistered control family in fitting')
        if row['label'] != int(row['family'] in {'P1', 'P1w'}):
            raise ValueError('Control label/family mismatch')
        rows.append(row)
    # Validate explicit attempted pair identities, including retained failures.
    by_source = {}
    for row in rows:
        by_source.setdefault(row['pair_id'], []).append(row)
    for identity, group in by_source.items():
        if len({row['family'] for row in group}) != len(group):
            raise ValueError('Duplicate control family for ' + identity)
        if not {'S0', 'P1', 'S_world', 'P1w'} <= {row['family'] for row in group}:
            raise ValueError('Missing mandatory attempted control family for ' + identity)
        if len({(row['word_id'], row['voice'], row['phone_collision_group']) for row in group}) != 1:
            raise ValueError('One pair identity mixes lexical/voice/group identity')
    return rows, rejected, {'directory': str(directory), 'items_sha256': sha(directory / 'items.jsonl'),
                            'content_sha256': sha(content), 'runtime': json.loads((directory / 'runtime.json').read_text())}


def arrays(rows):
    if any(row['status'] != 'measured' or row['features'] is None for row in rows):
        raise ValueError('Only explicitly measured rows may enter a fit matrix')
    return np.stack([row['features'] for row in rows]), np.array([row['label'] for row in rows], float)
