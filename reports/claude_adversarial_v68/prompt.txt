from __future__ import annotations

from pathlib import Path

import numpy as np
import torch
from torch import nn

from hindi_detector.corrected_synthesis import sha
from hindi_detector.intrinsic_vowels import FEATURE_NAMES as THIRD_NAMES
from hindi_detector.paired_local_vowels import FEATURE_NAMES as PAIRED_NAMES

QUERY_INDICES = [i for i, name in enumerate(PAIRED_NAMES)
                 if name.startswith('query_') or name in {'positive_margin_fraction', 'is_vowel_deletion', 'is_vowel_quality'}]
QUERY_NAMES = [prefix + PAIRED_NAMES[i] for prefix in ('primary_', 'secondary_') for i in QUERY_INDICES]
MODES = {'dual46': QUERY_NAMES, 'third33': ['independent_' + name for name in THIRD_NAMES] + ['independent_available'],
         'combined79': QUERY_NAMES + ['independent_' + name for name in THIRD_NAMES] + ['independent_available']}
if len(QUERY_INDICES) != 23 or any('reference' in name or 'delta_' in name for name in QUERY_NAMES):
    raise ValueError('Reference-free feature schema unexpectedly contains reference inputs')


def inputs(paired, third, available, mode):
    x, other, valid = np.asarray(paired, np.float32), np.asarray(third, np.float32), np.asarray(available, bool)
    if x.ndim != 2 or x.shape[1] != 114 or other.shape != (len(x), 32) or valid.shape != (len(x),):
        raise ValueError('Query-only input shapes differ from explicit measured schemas')
    if not np.isfinite(x).all() or not np.isfinite(other).all():
        raise ValueError('Query-only input values must be finite')
    query = x[:, QUERY_INDICES + [i + 57 for i in QUERY_INDICES]]
    independent = np.column_stack([np.where(valid[:, None], other, 0), valid.astype(np.float32)])
    if mode == 'dual46':
        return query
    if mode == 'third33':
        return independent
    if mode == 'combined79':
        return np.column_stack([query, independent])
    raise ValueError('Unknown query-only feature mode')


class QueryDeletion(nn.Module):
    def __init__(self, mode, architecture):
        super().__init__()
        if mode not in MODES:
            raise ValueError('Unknown query-only feature mode')
        width = len(MODES[mode])
        if architecture == 'linear':
            self.head = nn.Linear(width, 1)
        elif architecture == 'mlp8':
            self.head = nn.Sequential(nn.Linear(width, 8), nn.Tanh(), nn.Linear(8, 1))
        else:
            raise ValueError('Unknown query-only deletion architecture')

    def forward(self, values):
        return self.head(values).squeeze(-1)


def score(network, state, paired, third, available):
    values = inputs(paired, third, available, state['feature_mode'])
    normalized = np.clip((values - state['mean']) / state['scale'], -8, 8).astype(np.float32)
    with torch.inference_mode():
        return torch.sigmoid(network(torch.from_numpy(normalized))).cpu().numpy()


def load(path: Path):
    state = torch.load(path, map_location='cpu', weights_only=False)
    if state['schema'] != 'query_full_deletion_v66' or state['feature_names'] != MODES[state['feature_mode']]:
        raise ValueError('Query-only deletion checkpoint schema changed')
    root = Path(__file__).resolve().parents[1]
    for name, digest in state['code_sha256'].items():
        if sha(root / name) != digest:
            raise ValueError('Query-only checkpoint code changed: ' + name)
    network = QueryDeletion(state['feature_mode'], state['architecture'])
    network.load_state_dict(state['weights'], strict=True)
    return network.eval(), state

def represent(token, vocabulary):
    token = re.sub(r'[ˈˌ]', '', token)
    choices = REPRESENTATIONS.get(token, {token})
    ids = frozenset(vocabulary[t] for t in choices if t in vocabulary and vocabulary[t] > 3)
    if not ids:
        raise ValueError('No declared acoustic representation for ' + repr(token))
    return ids


def units_for_project(phones, ipa, vocabulary, *, any_schwa=False):
    all_vowels = frozenset(vocabulary[t] for t in VOWELS if t in vocabulary)
    units, projections = [], []
    for phone in phones:
        if phone == '.':
            continue
        if phone not in ipa or not ipa[phone]:
            raise ValueError('No lexical IPA for project phone ' + repr(phone))
        unit = all_vowels if any_schwa and phone == 'ax' else represent(ipa[phone], vocabulary)
        units.append(unit)
        projections.append({'project_phone': phone, 'ipa': ipa[phone],
                            'acoustic_tokens': sorted(t for t, i in vocabulary.items() if i in unit),
                            'any_vowel_site_presence_only': any_schwa and phone == 'ax'})
    return units, projections


def path_score(log_probs, units, *, trace=False):
    """CTC recurrence over explicit label sets, with optional maximum-path spans.

    Overlapping acoustic sets are a representation relaxation. Margins are
    uncalibrated evidence and must be checked against recorded native variation.
    """
    values = validate_log_probs(log_probs, 0)
    if not units or any(not u or 0 in u or min(u) < 0 or max(u) >= values.shape[1] for u in units):
        raise ValueError('Nonempty in-vocabulary nonblank unit sets required')
    states = [None]
    for unit in units:
        states.extend([frozenset(unit), None])
    emissions = np.column_stack([values[:, 0] if u is None else np.logaddexp.reduce(values[:, sorted(u)], axis=1) for u in states])
    skip = np.array([i > 1 and states[i] is not None and states[i] != states[i-2] for i in range(len(states))])
    alpha = np.full(len(states), -np.inf)
    alpha[:2] = emissions[0, :2]
    best = alpha.copy()
    pointers = np.zeros((len(values), len(states)), np.int8) if trace else None
    for frame in range(1, len(values)):
        one = np.r_[-np.inf, alpha[:-1]]
        two = np.r_[-np.inf, -np.inf, alpha[:-2]]
        two[~skip] = -np.inf
        alpha = np.logaddexp(np.logaddexp(alpha, one), two) + emissions[frame]
        if trace:
            choices = np.stack([best, np.r_[-np.inf, best[:-1]], np.where(skip, np.r_[-np.inf, -np.inf, best[:-2]], -np.inf)])
            pointers[frame] = choices.argmax(0)
            best = choices.max(0) + emissions[frame]
    likelihood = float(np.logaddexp(alpha[-1], alpha[-2]))
    if not trace:
        return likelihood
    if not math.isfinite(likelihood):
        raise ValueError('Insufficient phone frames for hypothesis')
    state = len(states) - 1 if best[-1] >= best[-2] else len(states) - 2
    route = np.empty(len(values), np.int32)
    for frame in range(len(values)-1, -1, -1):
        route[frame] = state
        if frame:
            state -= int(pointers[frame, state])
    spans = []
    for index in range(len(units)):
        frames = np.flatnonzero(route == 2*index + 1)
        if not len(frames):
            raise ValueError('Maximum path skipped a required unit')
        spans.append({'unit_index': index, 'start_frame': int(frames[0]), 'end_frame': int(frames[-1]+1),
                      'frames': len(frames), 'mean_log_emission': float(emissions[frames, 2*index+1].mean())})
    return {'log_score': likelihood, 'viterbi_score': float(max(best[-1], best[-2])), 'spans': spans}

def native_observed(f):return f[2]<0 and f[8]>-1.5 and f[30]==1


def full_gap(item,ordinal,compiled,variant,contrast,evidence,alternative=False):
    crops=windows(item,ordinal);native,other,at=make_paths(compiled,variant,contrast,evidence['retained_native_indices'])
    units=other if alternative else native;trace=path_score(crops[1][2],units,trace=True)
    if at and units[at-1]<=compiled['any_vowel'] or at+1<len(units) and units[at+1]<=compiled['any_vowel']:
        raise ValueError('Adjacent-vowel sequence needs a separate full-region segmentation')
    left=crops[1][1]+trace['spans'][at-1]['end_frame'] if at else crops[0][1]
    right=crops[1][1]+trace['spans'][at+1]['start_frame'] if at+1<len(units) else crops[0][1]+len(crops[0][2])
    start=max(crops[0][1],left);end=min(crops[0][1]+len(crops[0][2]),right)
    dt=item['detail']['phone_frame_seconds'];return int(start*dt*16000),int(end*dt*16000)
