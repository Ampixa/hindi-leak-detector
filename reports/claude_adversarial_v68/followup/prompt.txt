## 3. Enforcing focal-vowel ownership without filtering away real deletions

**Principle: eligibility may depend on text and pre-edit audio only.** The criteria are:
- the canonical G2P structure: non-vowel neighbours on both sides, a site the representation supports, no set overlap after the P0-7 fix;
- source/reference qualification on the unedited audio.

**Nothing that reads the query audio may exclude a positive.** Every query-side gate (ASR content, measurement failure, third-encoder availability) becomes an abstain outcome counted in the denominator. Report coverage as positions eligible out of positions requested, by context class.

**Make the evidence local:**
1. **Anchored-window comparison.** Compute the native Viterbi alignment once, freeze both anchor consonants, and score both hypotheses only on the frames from the left-anchor start to the right-anchor end. Differences elsewhere in the word then cancel. Use this local margin instead of whole-crop margins.
2. **Ownership statistic.** From forward-backward over the native path, measure how much of the focal state's posterior occupancy falls on frames whose best path belongs to a neighbouring unit. High borrowing means low ownership. Strong focal emission sitting on a neighbour's frames is not evidence that the focal vowel is present, which is the आइसके trap in either direction.
3. **Adjacent vowels.** Abstain by text-level rule and report it as a coverage gap. Do not exclude by measured outcome.
4. **Break the boundary circularity.** `full_gap` defines the removed region from the same primary encoder that later scores the features. Add a positive family whose boundaries are jittered by ±1–2 frames inside the owned interval, and ideally one whose boundaries come from the third encoder. Detection should be stable across them.

"""Exact CTC marginal over explicit sets of allowed labels at each phone.

Unlike the legacy category-state relaxation, the chosen token stays fixed
inside one phone state. Adjacent phones may skip a blank only when their
actual emitted token IDs differ, including when the allowed sets overlap.
This is versioned separately so existing checkpoints retain their features.
"""
from __future__ import annotations

import math

import numpy as np

from hindi_detector.ctc import validate_log_probs


def _without_each(values):
    left = np.r_[-np.inf, np.logaddexp.accumulate(values)[:-1]]
    right = np.r_[np.logaddexp.accumulate(values[::-1])[::-1][1:], -np.inf]
    return np.logaddexp(left, right)


def path_score(log_probs, units, *, trace=False):
    values = validate_log_probs(log_probs, 0)
    labels, positions, blanks = [], [], [0]
    state_labels = [0]
    for unit in units:
        if not unit or 0 in unit or min(unit) < 0 or max(unit) >= values.shape[1]:
            raise ValueError('Phone alternatives must be nonempty in-vocabulary nonblank IDs')
        ids = np.array(sorted(unit), np.int64)
        at = np.arange(len(state_labels), len(state_labels) + len(ids))
        labels.append(ids)
        positions.append(at)
        state_labels.extend(ids.tolist())
        blanks.append(len(state_labels))
        state_labels.append(0)
    if not units:
        value = float(values[:, 0].sum())
        return {'log_score': value, 'viterbi_score': value, 'spans': []} if trace else value
    matches = []
    for i, ids in enumerate(labels):
        lookup = {} if i == 0 else {token: j for j, token in enumerate(labels[i - 1])}
        matches.append(np.array([lookup.get(token, -1) for token in ids], np.int64))
    alpha = np.full(len(state_labels), -np.inf)
    alpha[0] = values[0, 0]
    alpha[positions[0]] = values[0, labels[0]]
    best = alpha.copy()
    pointers = np.zeros((len(values), len(state_labels)), np.int32) if trace else None
    for frame in range(1, len(values)):
        nxt = np.full_like(alpha, -np.inf)
        next_best = np.full_like(alpha, -np.inf) if trace else None
        nxt[0] = alpha[0] + values[frame, 0]
        if trace:
            next_best[0] = best[0] + values[frame, 0]
        for i, (ids, states) in enumerate(zip(labels, positions, strict=True)):
            previous_blank, following_blank = blanks[i], blanks[i + 1]
            skip = np.full(len(ids), -np.inf)
            if i:
                preceding = alpha[positions[i - 1]]
                skip[:] = np.logaddexp.reduce(preceding)
                same = matches[i] >= 0
                skip[same] = _without_each(preceding)[matches[i][same]]
            nxt[states] = np.logaddexp(np.logaddexp(alpha[states], alpha[previous_blank]), skip) + values[frame, ids]
            nxt[following_blank] = np.logaddexp(alpha[following_blank], np.logaddexp.reduce(alpha[states])) + values[frame, 0]
            if trace:
                skip_best = np.full(len(ids), -np.inf)
                skip_parent = np.zeros(len(ids), np.int32)
                if i:
                    prev_states = positions[i - 1]
                    prev_scores = best[prev_states]
                    top = int(np.argmax(prev_scores))
                    second_scores = prev_scores.copy()
                    second_scores[top] = -np.inf
                    second = int(np.argmax(second_scores))
                    index = np.where(matches[i] == top, second, top)
                    skip_best = prev_scores[index].copy()
                    if len(prev_states) == 1:
                        skip_best[matches[i] == top] = -np.inf
                    skip_parent = prev_states[index]
                choices = np.stack([best[states], np.full(len(ids), best[previous_blank]), skip_best])
                choice = choices.argmax(0)
                next_best[states] = choices.max(0) + values[frame, ids]
                parents = np.stack([states, np.full(len(ids), previous_blank), skip_parent])
                pointers[frame, states] = parents[choice, np.arange(len(ids))]
                top_state = states[int(np.argmax(best[states]))]
                parent = following_blank if best[following_blank] >= best[top_state] else top_state
                next_best[following_blank] = best[parent] + values[frame, 0]
                pointers[frame, following_blank] = parent
        alpha = nxt
        if trace:
            best = next_best
    endings = np.r_[positions[-1], blanks[-1]]
    likelihood = float(np.logaddexp.reduce(alpha[endings]))
    if not trace:
        return likelihood
    if not math.isfinite(likelihood):
        raise ValueError('No feasible exact CTC path for the declared phone alternatives')
    state = int(endings[np.argmax(best[endings])])
    maximum = float(best[state])
    route = np.zeros(len(values), np.int32)
    for frame in range(len(values) - 1, -1, -1):
        route[frame] = state
        if frame:
            state = int(pointers[frame, state])
    spans = []
    for i, states in enumerate(positions):
        frames = np.flatnonzero(np.isin(route, states))
        if not len(frames):
            raise ValueError('Exact maximum path omitted a required phone')
        emitted = {state_labels[int(route[frame])] for frame in frames}
        if len(emitted) != 1:
            raise AssertionError('One exact CTC phone changed token identity without a phone transition')
        token = emitted.pop()
        spans.append({'unit_index': i, 'start_frame': int(frames[0]), 'end_frame': int(frames[-1] + 1),
                      'frames': len(frames), 'token_id': token,
                      'mean_log_emission': float(values[frames, token].mean())})
    return {'log_score': likelihood, 'viterbi_score': maximum, 'spans': spans}

--- ACTUAL FILE tests/test_exact_phone_ctc.py ---
from __future__ import annotations

from itertools import product
import unittest

import numpy as np

from hindi_detector.ctc import log_likelihood
from hindi_detector.exact_phone_ctc import path_score


class ExactAllowedPhoneCTC(unittest.TestCase):
    def test_marginal_equals_enumeration_of_concrete_ctc_sequences(self):
        rng = np.random.default_rng(69)
        for units in [[{1, 2}, {2, 3}], [{1, 2}, {1, 2}], [{1}, {2, 3}, {3}], []]:
            for frames in (2, 3, 5):
                probabilities = rng.dirichlet(np.ones(4), size=frames)
                values = np.log(probabilities)
                terms = [log_likelihood(values, list(sequence)) for sequence in product(*units)]
                expected = float(np.logaddexp.reduce(terms))
                self.assertAlmostEqual(path_score(values, units), expected, places=11)

    def test_shared_token_requires_blank_but_different_tokens_can_skip(self):
        same = np.log(np.array([[.001, .001, .997, .001], [.001, .001, .997, .001]]))
        self.assertLess(path_score(same, [{1, 2}, {2, 3}]), -5)
        distinct = np.log(np.array([[.001, .997, .001, .001], [.001, .001, .001, .997]]))
        self.assertGreater(path_score(distinct, [{1, 2}, {2, 3}]), -.02)

    def test_one_phone_cannot_change_token_mid_state(self):
        values = np.log(np.array([[.001, .998, .001], [.001, .001, .998]]))
        self.assertLess(path_score(values, [{1, 2}]), -5)

    def test_trace_matches_exhaustive_viterbi(self):
        rng = np.random.default_rng(70)
        values = np.log(rng.dirichlet(np.ones(4), size=5))
        units = [{1, 2}, {2, 3}]
        candidates = []
        for route in product(range(4), repeat=5):
            decoded = [x for i, x in enumerate(route) if x and (i == 0 or x != route[i - 1])]
            if len(decoded) == len(units) and all(x in u for x, u in zip(decoded, units, strict=True)):
                candidates.append(sum(values[i, token] for i, token in enumerate(route)))
        trace = path_score(values, units, trace=True)
        self.assertAlmostEqual(trace['viterbi_score'], max(candidates), places=11)
        self.assertEqual(len(trace['spans']), 2)
        self.assertLessEqual(trace['spans'][0]['end_frame'], trace['spans'][1]['start_frame'])

    def test_impossible_trace_explicit_and_blank_only_supported(self):
        values = np.log(np.array([[.1, .9]]))
        self.assertTrue(np.isneginf(path_score(values, [{1}, {1}])))
        with self.assertRaises(ValueError):
            path_score(values, [{1}, {1}], trace=True)
        self.assertAlmostEqual(path_score(values, []), np.log(.1))

"""Reference-free internal-vowel evidence inside fixed consonant anchors.

The deletion path locates both anchors once per boundary recipe. Vowel and
blank-only hypotheses then use the IDENTICAL intervening frames; unrelated
vowels cannot improve that local likelihood. No word length, lexical identity,
speaker identity, source identity or reference-derived feature is emitted.
"""
from __future__ import annotations

import math

import numpy as np

from hindi_detector.exact_phone_ctc import path_score

FEATURE_NAMES = ['gap_deletion_margin_mean', 'gap_deletion_margin_min', 'gap_deletion_margin_max',
    'gap_vowel_mass_mean', 'gap_vowel_peak_mean', 'focal_vowel_log_emission_mean',
    'focal_ownership_mean', 'left_anchor_log_emission_mean', 'right_anchor_log_emission_mean',
    'gap_native_fit_mean', 'gap_blank_fit_mean', 'gap_available_fraction']
FLOOR = -30.


def _paths(compiled, variant, contrast, retained):
    at = retained.index(contrast['index'])
    native = [variant['units'][i] for i in retained]
    native[at] = compiled['any_vowel']
    return native, native[:at] + native[at + 1:], at


def measure(compiled, variant, contrast, crops):
    if contrast['kind'] != 'vowel_deletion' or len(crops) != 3:
        raise ValueError('Anchored evidence is only for deletion at three fixed word bounds')
    index, units, vowels = contrast['index'], variant['units'], compiled['any_vowel']
    if not 0 < index < len(units) - 1 or units[index - 1] & vowels or units[index + 1] & vowels:
        raise ValueError('Internal focal vowel requires two nonsyllabic text-level neighbors')
    retained = list(range(len(units)))
    removable = {i for i, phone in enumerate(variant['phones']) if i != index and phone in {'ax', 'h'}}
    def fit(indices):
        a, b, _ = _paths(compiled, variant, contrast, indices)
        return max(path_score(crops[1][2], a), path_score(crops[1][2], b))
    current = fit(retained)
    while removable:
        trials = []
        for at in sorted(removable):
            proposal = [i for i in retained if i != at]
            # Never remove either immediate consonant anchor around the focal
            # site, even when it happens to be optional /h/ in other positions.
            focal = proposal.index(index)
            if not 0 < focal < len(proposal) - 1 or proposal[focal - 1] != index - 1 or proposal[focal + 1] != index + 1:
                continue
            trials.append((fit(proposal), at, proposal))
        if not trials:
            break
        value, at, proposal = max(trials, key=lambda row: row[0])
        if value <= current + .01:
            break
        retained, current = proposal, value
        removable.remove(at)
    native, alternate, focal = _paths(compiled, variant, contrast, retained)
    measurements = []
    for padding, start_frame, values in crops:
        deletion = path_score(values, alternate, trace=True)
        left, right = deletion['spans'][focal - 1], deletion['spans'][focal]
        start, end = left['end_frame'], right['start_frame']
        if not 0 <= start <= end <= len(values):
            raise ValueError('Consonant anchors do not define an ordered local gap')
        gap = values[start:end]
        mass = peak = 0.
        if len(gap):
            native_score = path_score(gap, [vowels])
            blank_score = path_score(gap, [])
            if not math.isfinite(blank_score):
                raise ValueError('Nonfinite blank-only local hypothesis')
            if not math.isfinite(native_score):
                native_score = FLOOR * len(gap)
            p = np.exp(gap)
            vowel = p[:, sorted(vowels)].sum(1)
            nonblank = p[:, 4:].sum()
            mass = float(vowel.sum() / max(1e-12, nonblank))
            peak = float(vowel.max())
            native_fit, blank_fit = native_score / len(gap), blank_score / len(gap)
        else:
            # A zero-frame observed gap is evidence, not a measurement failure:
            # the two consonant peaks are adjacent. Represent the impossible
            # intervening-vowel hypothesis explicitly with the fixed censor.
            native_fit, blank_fit = FLOOR, 0.
        focal_log, ownership, span = FLOOR, 0., None
        try:
            intact = path_score(values, native, trace=True)
            span = intact['spans'][focal]
            focal_log = span['mean_log_emission']
            overlap = max(0, min(end, span['end_frame']) - max(start, span['start_frame']))
            ownership = overlap / max(1, span['end_frame'] - span['start_frame'])
        except ValueError as error:
            # No intact path is an explicit observed hypothesis outcome, not an
            # omitted positive. The deletion path above must still be feasible.
            if 'No feasible exact CTC path' not in str(error):
                raise
        measurements.append({'padding_seconds': padding, 'start_frame': start_frame,
            'gap_start_frame': start_frame + start, 'gap_end_frame': start_frame + end,
            'gap_deletion_margin': blank_fit - native_fit, 'gap_vowel_mass': mass, 'gap_vowel_peak': peak,
            'native_focal_log_emission': focal_log, 'focal_ownership': ownership,
            'left_anchor': left, 'right_anchor': right, 'native_focal_span': span,
            'gap_native_fit': native_fit, 'gap_blank_fit': blank_fit, 'gap_available': bool(len(gap))})
    def mean(name):
        return float(np.mean([row[name] for row in measurements]))
    margins = [row['gap_deletion_margin'] for row in measurements]
    features = np.array([np.mean(margins), min(margins), max(margins), mean('gap_vowel_mass'), mean('gap_vowel_peak'),
        mean('native_focal_log_emission'), mean('focal_ownership'),
        np.mean([row['left_anchor']['mean_log_emission'] for row in measurements]),
        np.mean([row['right_anchor']['mean_log_emission'] for row in measurements]),
        mean('gap_native_fit'), mean('gap_blank_fit'), mean('gap_available')], np.float32)
    if features.shape != (len(FEATURE_NAMES),) or not np.isfinite(features).all():
        raise ValueError('Invalid anchored-vowel evidence features')
    return features, {'windows': measurements, 'retained_native_indices': retained,
                      'native_target_phones': variant['phones'], 'no_reference_audio_used': True,
                      'likelihood': 'Exact allowed-token CTC inside identical fixed deletion-anchor gaps'}

class AnchoredVowelEvidence(unittest.TestCase):
    def evidence(self, tokens):
        p = np.full((len(tokens), 9), .0001)
        for i, token in enumerate(tokens):
            p[i, token] = 1 - .0001 * 8
        values = np.log(p)
        compiled = {'any_vowel': frozenset({7, 8})}
        variant = {'phones': ['k', 'ax', 'p', 'i', 't'],
                   'units': [{4}, {7}, {5}, {8}, {6}]}
        contrast = {'kind': 'vowel_deletion', 'index': 1}
        return measure(compiled, variant, contrast, [(pad, 0, values) for pad in (0., .03, .06)])

    def test_missing_vowel_at_equal_duration_changes_local_evidence(self):
        intact, detail = self.evidence([0, 4, 0, 7, 7, 0, 5, 0, 8, 0, 6, 0])
        missing, _ = self.evidence([0, 4, 0, 0, 0, 0, 5, 0, 8, 0, 6, 0])
        self.assertLess(intact[0], 0)
        self.assertGreater(missing[0], 0)
        self.assertGreater(intact[3], missing[3])
        self.assertGreater(intact[6], .9)
        self.assertTrue(detail['no_reference_audio_used'])

    def test_neighbor_vowel_does_not_fill_the_focal_gap(self):

        b, _ = self.evidence([0, 4, 0, 0, 0, 0, 5, 8, 8, 8, 6, 0])

"""Retain every attempted post-edit outcome, including explicit abstentions."""
from __future__ import annotations

import logging


def attempt(record, measurement):
    try:
        result = measurement()
    except ValueError as error:
        logging.warning('Retained post-edit abstention %s/%s: %s', record['pair_id'], record['family'], error)
        return {**record, 'status': 'abstain', 'features': None, 'reason': str(error)}
    return {**record, 'status': 'measured', **result}

def world_versions(samples, start, end):
    """Matched WORLD reconstruction and half-duration focal vowel, pitch held."""
    signal = np.asarray(samples, np.float64)
    f0, times = pyworld.dio(signal, 16000, f0_floor=50., f0_ceil=650., frame_period=5.)
    f0 = pyworld.stonemask(signal, f0, times, 16000)
    spectrum = pyworld.cheaptrick(signal, f0, times, 16000)
    noise = pyworld.d4c(signal, f0, times, 16000)
    original = pyworld.synthesize(f0, spectrum, noise, 16000, frame_period=5.)
    if len(original) < len(samples):
        raise ValueError('WORLD reconstruction shorter than declared input')
    replacement = (end - start) // 2
    target_samples = len(samples) - (end - start) + replacement
    source_knots = np.array([0., start / 16000, end / 16000, len(samples) / 16000])
    target_knots = np.array([0., start / 16000, (start + replacement) / 16000, target_samples / 16000])
    target_times = np.minimum(np.arange(int(np.ceil(target_samples / 80)) + 1) * .005, target_samples / 16000)
    query = np.interp(target_times, target_knots, source_knots)
    query = np.clip(query, times[0], times[-1])
    nearest = np.clip(np.searchsorted(times, query), 1, len(times) - 1)
    nearest -= np.abs(times[nearest - 1] - query) < np.abs(times[nearest] - query)
    sp = np.ascontiguousarray(interp1d(times, spectrum, axis=0)(query))
    ap = np.ascontiguousarray(interp1d(times, noise, axis=0)(query))
    compressed = pyworld.synthesize(np.ascontiguousarray(f0[nearest]), sp, ap, 16000, frame_period=5.)
    if len(compressed) < target_samples:
        raise ValueError('WORLD compression shorter than declared time map')
    return original[:len(samples)].astype(np.float32), compressed[:target_samples].astype(np.float32), replacement

def sha(path: Path) -> str:
    digest = hashlib.sha256()
    with path.open('rb') as handle:
        for block in iter(lambda: handle.read(8 * 1024**2), b''):
            digest.update(block)
    return digest.hexdigest()


def require_pair(pair: dict) -> None:
    if pair.get('schema') != 'teacher_indicf5_pair_v14' or set(pair.get('systems', {})) != set(SYSTEMS):
        raise ValueError('Only the user-requested teacher/IndicF5 pair is accepted; no legacy fallback')
    teacher, f5 = (pair['systems'][s] for s in SYSTEMS)
    if teacher['layers'] != 24 or teacher['step'] != 250000 or f5['repo'] != 'ai4bharat/IndicF5':
        raise ValueError('Wrong teacher architecture/step or IndicF5 source')
    if teacher['sha256'] != '2b4636e3aed0e4f2d035630e504be95fc58f494b690a851a91bfe0dd8299516f' or f5['sha256'] != 'ba7f3671180fb7784e24bd1dafc96e729a38ce02e7f6d3877cdef32525a1865c':
        raise ValueError('Unapproved checkpoint bytes')

"""Pinned research-only Allosaurus encoder; never a native-pronunciation oracle."""
from __future__ import annotations

from argparse import Namespace
import importlib
import json
from pathlib import Path
import sys

import numpy as np
import torch

from hindi_detector.corrected_synthesis import sha
from hindi_detector.ctc import validate_log_probs


class IndependentPhone:
    def __init__(self, root: Path):
        directory = root / 'runtime/independent_phone_v64'
        inventory = json.loads((directory / 'inventory.json').read_text())
        protected = [row for row in inventory if row['path'].startswith(('source/', 'models/'))]
        for row in protected:
            path = (directory / row['path']).resolve(strict=True)
            if not path.is_relative_to(directory.resolve()) or sha(path) != row['sha256']:
                raise ValueError('Pinned independent encoder source/model bytes changed: ' + row['path'])
        sources = [p for p in (directory / 'source').iterdir() if (p / 'allosaurus').is_dir()]
        if len(sources) != 1:
            raise ValueError('Independent encoder source root is ambiguous')
        sys.dont_write_bytecode = True
        sys.path[:0] = [str(directory / 'deps'), str(sources[0])]
        package = importlib.import_module('allosaurus')
        if not Path(package.__file__).resolve().is_relative_to(sources[0].resolve()):
            raise ValueError('Unexpected globally installed Allosaurus package')
        pm_module = importlib.import_module('allosaurus.pm.factory')
        am_module = importlib.import_module('allosaurus.am.factory')
        self.Audio = importlib.import_module('allosaurus.audio').Audio
        read_unit = importlib.import_module('allosaurus.lm.unit').read_unit
        model = directory / 'models/uni2005'
        config = Namespace(model='uni2005', device_id=-1, lang='ipa', approximate=False, prior=None)
        self.pm = pm_module.read_pm(model, config)
        self.am = am_module.read_am(model, config).eval()
        units = read_unit(model / 'phone.txt').unit_to_id
        self.original_vocabulary = units
        if sorted(units.values()) != list(range(self.am.phone_size)) or units['<blk>'] != 0:
            raise ValueError('Observed independent phone inventory/model output mismatch')
        # Project measurement code reserves IDs1..3. Add exactly zero-probability
        # columns, preserving every original phone/blank probability and ID map.
        self.vocabulary = {'<blk>': 0, '<audit_reserved1>': 1, '<audit_reserved2>': 2, '<audit_reserved3>': 3,
                           **{token: index + 3 for token, index in units.items() if index}}
        self.frame_seconds = self.pm.config.window_shift * self.pm.config.feature_window
        self.metadata = {'encoder': 'Allosaurus uni2005', 'source_revision': 'd9f1adaf47d3b3765b41f4177da62a051516d636',
                         'model_sha256': sha(model / 'model.pt'), 'vocabulary': self.vocabulary,
                         'frame_seconds': self.frame_seconds, 'dither_seed': 6400,
                         'language_inventory_mask': None, 'native_phone_accuracy': None,
                         'licence': 'Source GPL-3.0; checkpoint-specific licence not supplied in release; no redistribution.'}

    def encode(self, samples, rate=16000):
        x = np.asarray(samples)
        if x.ndim != 1 or len(x) < rate * .1 or not np.isfinite(x).all() or rate <= 0:
            raise ValueError('Independent phone encoder requires finite mono audio of at least100ms')
        if np.max(np.abs(x)) > 1.001:
            raise ValueError('Expected normalized floating-point audio, not int16 magnitudes')
        np.random.seed(6400)
        signal = np.round(np.clip(x, -1, 1) * 32767).astype(np.int16)
        features = self.pm.compute(self.Audio(signal, rate))
        if features.ndim != 2 or not np.isfinite(features).all():
            raise ValueError('Independent encoder feature extraction failed')
        with torch.inference_mode():
            logits = self.am(torch.from_numpy(features[None]), torch.tensor([len(features)]))[0]
            values = logits.log_softmax(-1).cpu().numpy().astype(np.float64)
        padded = np.full((len(values), values.shape[1] + 3), -np.inf, np.float64)
        padded[:, 0] = values[:, 0]
        padded[:, 4:] = values[:, 1:]
        return validate_log_probs(padded, 0)

    def decode(self, values):
        inverse = {index: token for token, index in self.vocabulary.items()}
        ids = values.argmax(1)
        return [{'phone': inverse[int(index)], 'start': frame * self.frame_seconds,
                 'model_window_seconds': .045}
                for frame, index in enumerate(ids)
                if index > 3 and (frame == 0 or index != ids[frame - 1])]
